class Node:
    def __init__(self, name: str, *dependencies):
        self.name = name
        self.dependencies = dependencies 
						 // node that this node depens on (아래)
        self.value = None
        self.grad = None
    def forward(self):
        raise NotImplementedError
    def backward(self):
        raise NotImplementedError
    def grad_initialize(self):
        raise NotImplementedError
        
==================================================

forward -> 노드 자신의 value를 찾음 (값 구하기)

backward -> dependencies의 grad를 구함
		=dependencies에 대한 grad (파란 글씨 구하기)

grad_initalize -> 기울기 초기화(차원 맞추기)

class MatMul(Node):
   """Matrix multiplication"""
def forward(self):
x, y = self.dependencies

self.value = x.value @ y.value

def backward(self):
x, y = self.dependencies

x.grad += self.grad @ y.vlaude.T
y.grad += x.value.T @ self.grad

def grad_initialize(self):
self.grad = np.zeros_like(self.value)

def backward(self, pred, target):
pred_reshaped = pred.reshape(-1,1)
target_reshaped = target.reshape(-1,1)
		//reshape(n,m) : n=행의 크기, m=열의 크기

def update(self, gradients, learning_rate):
self.W1 -= learning_rate * gradients["W1"]
self.b1 -= learning_rate * gradients["b1"]
self.W2 -= learning_rate * gradients["W2"]
self.b2 -= learning_rate * gradients["b2"]

def train(self, X, Y, epochs=100, learning_rate=0.1):
for i in range(epochs):
for j in range(X.shape[0]):      <-행(Row)의 개수
x_sample = X[j:j+1]          
y_sample = Y[jj:j+1]           -> j번째 행
pred = self.forward(x_sample)
gradients = self.backward(pred, y_sample)
self.update(gradients, learning_rate)
loss = self.loss(pred, y_sample)
