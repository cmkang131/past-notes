def sparse_vector_dot_product(v1: SparseVector, v2: SparseVector) -> float:
if len(v1) > len(v2):
v1, v2 = v2, v1         //=> 항상 len(v2)>len(v1)
 
return sum(v * v2.get(k,0) for k,v in v1.items())
===================================================

def increment_sparse_vector(v1: SparseVector, scale: float, v2: SparseVector) -> None:
    if scale == 0:
      return
    for key, value in v2.items():
      v1[key] = v1[key] + scale * value

def find_most_frequent_words(text: str) -> Set[str]:
if not text:     //text가 없으면
  return set()
        
  counts = collections.defaultdict(int)
  for word in text.split():
  counts[word] += 1
        
  max_value = max(counts.values())
    
  # 딕셔너리의 아이템을 순회하며 최댓값과 일치하는 단어만 집합으로 묶어 반환
  return {word for word, count in counts.items() if count == max_value}
