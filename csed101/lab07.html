list comprehension(리스트 컴프리헨션)

zip()
  • 2개 이상의 list(리스트)를, 각 리스트의 같은 인덱스 원소끼리 묶은 tuple(튜플)을 원소로 하는 리스트 반환할 수 있음
  • list(zip(list1, list2)) 형태로 list안에 zip을 넣어야 함
L = list(zip([1,2,3], [4,5,6]))
print(L)
  • \>\>\> \[(1, 4), (2, 5), (3, 6)\]
L = [sum(x) for x in zip((1,2,3),(4,5,6),(7,8,9))]
print(L)
  • >>> [12, 15, 18]

'짝수 개수 구하기'

def count_even(*n):
    cnt = 0
    for v in n:
        if v % 2 == 0:
            cnt += 1
    return cnt

print(count_even(1,2,3,4,5,6,7,8,9))
실수 2 
'1개 이상의 3차원 벡터들을 전달받아 벡터들의 합을 구하여 반환'

def sum_vectors(*vectors):
    return [sum(vals) for vals in zip(*vectors)]

v1 = [0,1,2]
v2 = [0.5, 0.5, 0.5]
v3 = [1,3,5]

vector_sum = sum_vectors(v1,v2,v3)
print(vector_sum)

실습 3 
'숫자로 구성된 리스트 2개'
'=> 리스트 합친 후 정렬한 새로운 리스트 반환'

def merge_list(l1=[0],l2=[0]):
    ml = l1 + l2
    ml.sort()
    return ml


if __name__ == '__main__':
    l = [3, 5, 9, 1, 2]
    ml1 = merge_list(l,[2,1])
    ml2 = merge_list([6,9,4])
    ml3 = merge_list()
    print(ml1) # [1, 1, 2, 2, 3, 5, 9]
    print(ml2) # [0, 4, 6, 9]
    print(ml3) # [0, 0]
set(집합)
  • 순서 없이 항목 저장, 항목이 중볼될 수 없음

s = set([1,2,3,3,3,4,2])

#add

'1,3,4,5,7,9,10'

print(s)
'set()'
  • 집합 관련 내장 함수
s1 = {1,2,3}
s2 = {3,4,5}

#합집합(변경x,반환)
print(s1.union(s2))

실습 4 
'두 사람이 등산한 산의 목록이 각각 집합(set)으로 주어질 때, 두 사람이'
'모두 방문한 산과 한쪽만 방문한 산을 구하여 반환'

def compare_mountains(h1,h2):
    both = h1&h2
    one = (h1-h2)|(h2-h1)
    return both, one

if __name__ == '__main__':
    hiker_1 = {'설악산', '한라산', '지리산', '태백산'}
    hiker_2 = {'설악산', '주왕산', '덕유산', '한라산'}
    hiker_3 = {'무장산', '주왕산'}
    both1, one1 = compare_mountains(hiker_1, hiker_2)
    both2, one2 = compare_mountains(hiker_1, hiker_3)
    print(both1, one1)
    "{'한라산', '설악산'} {'지리산', '태백산', '덕유산', '주왕산'}"
    print(both2, one2)
    "set() {'한라산', '지리산', '태백산', '무장산', '설악산', '주왕산'}"
