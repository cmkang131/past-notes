Class(클래스) 
class Dog:
    count = 0 # => 클래스 변수 (self가 없음)

    def __init__(self,name,age):
        self.name = name
        self.age = age
        Dog.count += 1 # => 클래스 내에서 Dog.count로 호출
    
    def bark(self):
        print('war war')

dog1 = Dog('강아지',16)
dog1.bark()
Dog.bark(dog1)

dog2 = Dog('고양이', 17)
print(Dog.count) #-> 2
  • self.<statement> ⇒ Class variable(클래스 변수)
  • def <statement>(self, ): ⇒ Method(메서드)
  • <Instance(인스턴스)> = <classname>( ~ ) ⇒ 클래스로 인스턴트 생성
  • \<instance\>.\<method\>()
  • \<classname\>.\<method\>(\<instance\>)
    • 둘 다 메서드를 호출할 수 있음
실습 1
  • 은행 계좌 클래스 생성
'은행 계좌 클래스 만들기'


class BankAccount():
    def __init__(self, name=None, balance=0):
        self.name = name
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount

    def withdraw(self, amount):
        if balance >= amount:
            self.balance -= amount

    def get_info(self):
        print(f'name: {self.name}')
        print(f'balance: {self.balance}')

acc1 = BankAccount('이현승', 3000)
acc1.deposit(4000)
acc1.get_info()
