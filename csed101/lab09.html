Operator Overloading(연산자 오버로딩)
  • acc1 += 1000
  • > acc1 = acc1.iadd(1000)
  • 이런 식으로 작동함
    • return self
    • 를 이용해서 self를 반환해 줘야 함
실습 1 
'응행 계좌에서 Operator Overloading'

class BankAccount():
    def __init__(self, name=None, balance=0):
        self.name = name
        self.balance = balance

    def __add__(self, amount):
        self.balance += amount
        return self

    def __iadd__(self, amount):
        self.balance += amount
        return self

    def __sub__(self, amount):
        if self.balance >= amount:
            self.balance -= amount
            return self
        else:
            print('잔액 부족')
            return self

    def __isub__(self, amount):
        if self.balance >= amount:
            self.balance -= amount
            return self
        else:
            print('잔액 부족')
            return self

    def __str__(self):
        return f'name: {self.name}, balance: {self.balance}'

if __name__ == "__main__":
    acc1 = BankAccount("홍길동", 1000)

Method Overriding
  • 기존 클래스 메서드를 상속 클래스에서 재정의 하는 것
  • 기존 클래스의 메서드를 사용하고자 하는 경우 super() 사용
class A:
    def __init__(self):
        print("this is [A] class and [__init__] function")

class B(A):
    def __init__(self):
        super().__init__()
        print("this is [b] class and [__init__] function")

a = A()
b = B()

실습 1 상속 및 오버라이딩 [사람 ⇒ 학생]
  • 사람 → 학생 + sid(학번) 기능
class Person:

    def __init__(self, name):
        self.name = name

    def introduce(self):
        print(f'안녕하세요, 저는 {self.name}입니다.')

class Student(Person):

    def __init__(self, name, sid):
        super().__init__(name)   # 필요한 변수 입력해줘야 함
        self.sid = sid
    
    def introduce(self):
        super().introduce()
        print(f'학번: {self.sid}')


# 테스트 코드
p = Person("고길동")
p.introduce()
print()
s = Student("박희동", 20250001)
s.introduce()

실습 2 상속 및 오버라이딩 [사람 ⇒ 교수, 학생]
  • 지도교수 - 지도학생 introduce되는 기능 추가
class Person:

    def __init__(self, name):
        self.name = name

    def introduce(self):
        print(f'안녕하세요, 저는 {self.name}입니다.')

class Student(Person):

    def __init__(self, name, sid):
        super().__init__(name)   # 필요한 변수 입력해줘야 함
        self.sid = sid
        self.advisor = None 
    
    def introduce(self):
        super().introduce()
        print(f'학번: {self.sid}')
        if self.advisor:
            print(f'지도교수: {self.advisor}')

class Professor(Person):

    def __init__(self,name):
        super().__init__(name)
        self.students = []

    def add_student(self, stu):
        self.students.append(stu)
        stu.advisor = self.name

    def introduce(self):
        super().introduce()
        if self.students:
            names = ','.join(stu.name for stu in self.students)
            print(f'지도학생: {names}')
        else:
            print(f'지도학생: 없음')

# 테스트 코드
p1 = Professor('박교수')
p2 = Professor('홍교수')
s1 = Student('김학생', 20191234)
s2 = Student('이학생', 20221234)
p1.add_student(s1)
p1.add_student(s2)
s1.introduce()
s2.introduce()

p2.introduce()
